> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** construir y probar `src/core`. Es TypeScript sin DOM e incluye el modelo de cartas, el validador de contenido, el emparejamiento, las plantillas, el selector de cartas, las bolsas de turnos y modos, y el reducer de la partida.

```ts
export const KNOWN_VARIABLES = ['jugador', 'otro', 'compatible'] as const;
export type Variable = (typeof KNOWN_VARIABLES)[number];

const VARIABLE_PATTERN = /\{([^{}]*)\}/g;

/** Every `{name}` occurrence in order, including repeats and unknown names. */
export function extractVariableNames(text: string): string[] {
  return Array.from(text.matchAll(VARIABLE_PATTERN), (match) => match[1] ?? '');
}

export function isKnownVariable(name: string): name is Variable {
  return (KNOWN_VARIABLES as readonly string[]).includes(name);
}

export function replaceVariables(text: string, replacer: (name: string) => string): string {
  return text.replace(VARIABLE_PATTERN, (_match, name: string) => replacer(name));
}

/** True when a brace is left over after removing every well-formed `{name}`. */
export function hasStrayBraces(text: string): boolean {
  const withoutVariables = replaceVariables(text, () => '');
  return withoutVariables.includes('{') || withoutVariables.includes('}');
}
```

```ts
import {
  CARD_TYPES,
  CONTACT_TAG,
  GROUP_CARD_TYPES,
  LEVELS,
  type Card,
  type CardType,
  type Level,
  type Pack,
} from './types';
import { extractVariableNames, hasStrayBraces, isKnownVariable } from './variables';

export interface ContentIssue {
  readonly source: string;
  readonly cardId: string | null;
  readonly message: string;
}

export function formatIssue(issue: ContentIssue): string {
  return issue.cardId === null
    ? `${issue.source}: ${issue.message}`
    : `${issue.source} [${issue.cardId}]: ${issue.message}`;
}

export class ContentError extends Error {
  readonly issues: readonly ContentIssue[];

  constructor(issues: readonly ContentIssue[]) {
    super(`Invalid content (${issues.length} issue(s)):\n${issues.map(formatIssue).join('\n')}`);
    this.name = 'ContentError';
    this.issues = issues;
  }
}

export const ID_PATTERN = /^[a-z0-9]+(?:-[a-z0-9]+)*$/;
export const MAX_TIMER_SECONDS = 600;

const PACK_FIELDS = new Set(['id', 'version', 'lang', 'cards']);
const CARD_FIELDS = new Set(['id', 'type', 'level', 'text', 'timerSeconds', 'tags']);

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function isOneOf<T extends string>(options: readonly T[], value: unknown): value is T {
  return typeof value === 'string' && (options as readonly string[]).includes(value);
}

function textProblems(text: string, type: CardType | null, tags: readonly unknown[]): string[] {
  const problems: string[] = [];
  if (hasStrayBraces(text)) problems.push('text has unbalanced braces');
  const names = extractVariableNames(text);
  for (const name of new Set(names)) {
    if (!isKnownVariable(name)) problems.push(`unknown variable {${name}}`);
  }
  if (type !== null && GROUP_CARD_TYPES.includes(type) && names.length > 0) {
    problems.push(`${type} cards are for the whole group and cannot use player variables`);
  }
  if (tags.includes(CONTACT_TAG) && !names.includes('compatible')) {
    problems.push(`cards tagged "${CONTACT_TAG}" must use {compatible}`);
  }
  return problems;
}

export interface CardCheck {
  readonly card: Card | null;
  readonly label: string;
  readonly problems: readonly string[];
}

/** Validates one raw card. `label` identifies it when it has no usable id. */
export function checkCard(raw: unknown, label: string): CardCheck {
  if (!isRecord(raw)) return { card: null, label, problems: ['card must be a JSON object'] };

  const { id, type, level, text, tags, timerSeconds } = raw;
  const cardLabel = typeof id === 'string' && id.length > 0 ? id : label;
  const problems: string[] = [];

  for (const key of Object.keys(raw)) {
    if (!CARD_FIELDS.has(key)) problems.push(`unknown field "${key}"`);
  }
  if (typeof id !== 'string' || !ID_PATTERN.test(id)) {
    problems.push(`id must match ${ID_PATTERN}, got ${JSON.stringify(id)}`);
  }
  if (!isOneOf(CARD_TYPES, type)) {
    problems.push(`type must be one of ${CARD_TYPES.join(', ')}, got ${JSON.stringify(type)}`);
  }
  if (!isOneOf(LEVELS, level)) {
    problems.push(`level must be one of ${LEVELS.join(', ')}, got ${JSON.stringify(level)}`);
  }
  if (typeof text !== 'string' || text.trim().length === 0) {
    problems.push('text must be a non-empty string');
  }
  if (!Array.isArray(tags) || !tags.every((tag) => typeof tag === 'string' && tag.length > 0)) {
    problems.push('tags must be an array of non-empty strings');
  }
  if (timerSeconds !== undefined) {
    if (
      typeof timerSeconds !== 'number' ||
      !Number.isInteger(timerSeconds) ||
      timerSeconds < 1 ||
      timerSeconds > MAX_TIMER_SECONDS
    ) {
      problems.push(
        `timerSeconds must be an integer between 1 and ${MAX_TIMER_SECONDS}, got ${JSON.stringify(timerSeconds)}`,
      );
    }
    if (type !== 'DARE') problems.push('timerSeconds is only allowed on DARE cards');
  }
  if (typeof text === 'string') {
    problems.push(...textProblems(text, isOneOf(CARD_TYPES, type) ? type : null, Array.isArray(tags) ? tags : []));
  }

  if (problems.length > 0) return { card: null, label: cardLabel, problems };

  const card: Card = {
    id: id as string,
    type: type as CardType,
    level: level as Level,
    text: text as string,
    tags: [...(tags as string[])],
    ...(timerSeconds === undefined ? {} : { timerSeconds: timerSeconds as number }),
  };
  return { card, label: cardLabel, problems: [] };
}

/** Parses a raw pack. Throws ContentError listing every problem found. */
export function parsePack(raw: unknown, source: string): Pack {
  if (!isRecord(raw)) {
    throw new ContentError([{ source, cardId: null, message: 'pack must be a JSON object' }]);
  }

  const issues: ContentIssue[] = [];
  const packIssue = (message: string): void => {
    issues.push({ source, cardId: null, message });
  };

  for (const key of Object.keys(raw)) {
    if (!PACK_FIELDS.has(key)) packIssue(`unknown pack field "${key}"`);
  }
  const { id, version, lang, cards } = raw;
  if (typeof id !== 'string' || !ID_PATTERN.test(id)) {
    packIssue(`pack id must match ${ID_PATTERN}, got ${JSON.stringify(id)}`);
  }
  if (typeof version !== 'number' || !Number.isInteger(version) || version < 1) {
    packIssue(`version must be a positive integer, got ${JSON.stringify(version)}`);
  }
  if (typeof lang !== 'string' || lang.length === 0) {
    packIssue(`lang must be a non-empty string, got ${JSON.stringify(lang)}`);
  }
  if (!Array.isArray(cards)) {
    packIssue('cards must be an array');
    throw new ContentError(issues);
  }

  const parsed: Card[] = [];
  const seen = new Set<string>();
  cards.forEach((rawCard: unknown, index: number) => {
    const check = checkCard(rawCard, `#${index}`);
    for (const message of check.problems) issues.push({ source, cardId: check.label, message });
    if (check.card === null) return;
    if (seen.has(check.card.id)) {
      issues.push({ source, cardId: check.card.id, message: 'duplicate id in pack' });
      return;
    }
    seen.add(check.card.id);
    parsed.push(check.card);
  });

  if (issues.length > 0) throw new ContentError(issues);
  return { id: id as string, version: version as number, lang: lang as string, cards: parsed };
}
```

```ts
import { CARD_TYPES, LEVELS, type Card, type Pack } from './types';
import { ContentError, type ContentIssue } from './validator';
import { replaceVariables } from './variables';

export interface CorpusRules {
  readonly bannedSoftWords: readonly string[];
  readonly nearDuplicateThreshold: number;
}

/** Initial value; plan 3 calibrates it against real batches. */
export const DEFAULT_NEAR_DUPLICATE_THRESHOLD = 0.8;

/** Lowercase words without accents or punctuation; every variable becomes "var". */
export function normalizeWords(text: string): string[] {
  return replaceVariables(text, () => ' var ')
    .normalize('NFD')
    .replace(/\p{M}/gu, '')
    .toLowerCase()
    .replace(/[^\p{L}\p{N}]+/gu, ' ')
    .split(' ')
    .filter((word) => word.length > 0);
}

/** Word trigrams; texts shorter than three words become a single shingle. */
export function shingles(text: string): Set<string> {
  const words = normalizeWords(text);
  if (words.length < 3) return new Set([words.join(' ')]);
  const result = new Set<string>();
  for (let i = 0; i + 3 <= words.length; i++) result.add(words.slice(i, i + 3).join(' '));
  return result;
}

export function jaccard(a: ReadonlySet<string>, b: ReadonlySet<string>): number {
  if (a.size === 0 && b.size === 0) return 1;
  let shared = 0;
  for (const item of a) if (b.has(item)) shared++;
  return shared / (a.size + b.size - shared);
}

export interface NearDuplicate {
  readonly a: Card;
  readonly b: Card;
  readonly similarity: number;
}

/** Pairs of cards of the same type whose trigram similarity is >= threshold. */
export function findNearDuplicates(cards: readonly Card[], threshold: number): NearDuplicate[] {
  const prepared = cards.map((card) => ({ card, shingles: shingles(card.text) }));
  const result: NearDuplicate[] = [];
  for (let i = 0; i < prepared.length; i++) {
    const first = prepared[i];
    if (first === undefined) continue;
    for (let j = i + 1; j < prepared.length; j++) {
      const second = prepared[j];
      if (second === undefined || first.card.type !== second.card.type) continue;
      const similarity = jaccard(first.shingles, second.shingles);
      if (similarity >= threshold) result.push({ a: first.card, b: second.card, similarity });
    }
  }
  return result;
}

/** Banned entries (words or phrases) present as whole words in the card text. */
export function findBannedWords(card: Card, banned: readonly string[]): string[] {
  const haystack = ` ${normalizeWords(card.text).join(' ')} `;
  return banned.filter((entry) => {
    const needle = normalizeWords(entry).join(' ');
    return needle.length > 0 && haystack.includes(` ${needle} `);
  });
}

export function countByCombination(cards: readonly Card[]): Map<string, number> {
  const counts = new Map<string, number>();
  for (const type of CARD_TYPES) for (const level of LEVELS) counts.set(`${type}/${level}`, 0);
  for (const card of cards) {
    const key = `${card.type}/${card.level}`;
    counts.set(key, (counts.get(key) ?? 0) + 1);
  }
  return counts;
}

/** Cross-pack rules: unique ids, banned words in SOFT, near-duplicates. */
export function checkCorpus(packs: readonly Pack[], rules: CorpusRules): ContentIssue[] {
  const issues: ContentIssue[] = [];
  const sourceById = new Map<string, string>();
  const unique: Card[] = [];

  for (const pack of packs) {
    for (const card of pack.cards) {
      const owner = sourceById.get(card.id);
      if (owner !== undefined) {
        issues.push({ source: pack.id, cardId: card.id, message: `id already used in pack ${owner}` });
        continue;
      }
      sourceById.set(card.id, pack.id);
      unique.push(card);
    }
  }

  const sourceOf = (cardId: string): string => {
    const source = sourceById.get(cardId);
    if (source === undefined) throw new Error(`Card ${cardId} has no source pack`);
    return source;
  };

  for (const card of unique) {
    if (card.level !== 'SOFT') continue;
    const banned = findBannedWords(card, rules.bannedSoftWords);
    if (banned.length > 0) {
      issues.push({ source: sourceOf(card.id), cardId: card.id, message: `banned words in SOFT card: ${banned.join(', ')}` });
    }
  }

  for (const duplicate of findNearDuplicates(unique, rules.nearDuplicateThreshold)) {
    issues.push({
      source: sourceOf(duplicate.a.id),
      cardId: duplicate.a.id,
      message: `near-duplicate of ${duplicate.b.id} (similarity ${duplicate.similarity.toFixed(2)})`,
    });
  }

  return issues;
}

export function assertCorpus(packs: readonly Pack[], rules: CorpusRules): void {
  const issues = checkCorpus(packs, rules);
  if (issues.length > 0) throw new ContentError(issues);
}
```

```ts
export interface Rng {
  /** Uniform integer in [0, maxExclusive). Throws if maxExclusive is not a positive integer. */
  nextInt(maxExclusive: number): number;
}

function assertRange(maxExclusive: number): void {
  if (!Number.isInteger(maxExclusive) || maxExclusive < 1) {
    throw new RangeError(`maxExclusive must be a positive integer, got ${maxExclusive}`);
  }
}

/** Deterministic generator (mulberry32) for tests and reproducible games. */
export function createSeededRng(seed: number): Rng {
  let state = seed >>> 0;
  return {
    nextInt(maxExclusive: number): number {
      assertRange(maxExclusive);
      state = (state + 0x6d2b79f5) >>> 0;
      let t = state;
      t = Math.imul(t ^ (t >>> 15), t | 1);
      t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
      const unit = ((t ^ (t >>> 14)) >>> 0) / 4294967296;
      return Math.floor(unit * maxExclusive);
    },
  };
}

export function createMathRng(): Rng {
  return {
    nextInt(maxExclusive: number): number {
      assertRange(maxExclusive);
      return Math.floor(Math.random() * maxExclusive);
    },
  };
}

/** Fisher–Yates on a copy. */
export function shuffle<T>(items: readonly T[], rng: Rng): T[] {
  const result = [...items];
  for (let i = result.length - 1; i > 0; i--) {
    const j = rng.nextInt(i + 1);
    const current = result[i] as T;
    result[i] = result[j] as T;
    result[j] = current;
  }
  return result;
}

export function pickOne<T>(items: readonly T[], rng: Rng): T {
  if (items.length === 0) throw new Error('Cannot pick from an empty list');
  return items[rng.nextInt(items.length)] as T;
}
```

```ts
import { shuffle, type Rng } from './rng';

export interface BagDraw<T> {
  readonly item: T;
  readonly bag: readonly T[];
}

function refill<T>(items: readonly T[], last: T | null, rng: Rng): T[] {
  const bag = shuffle(items, rng);
  if (bag.length > 1 && bag[0] === last) {
    const swapWith = 1 + rng.nextInt(bag.length - 1);
    const first = bag[0] as T;
    bag[0] = bag[swapWith] as T;
    bag[swapWith] = first;
  }
  return bag;
}

/**
 * Draws the next item of a shuffle bag. Bag entries missing from `items` are dropped.
 * An empty bag is refilled with every item, shuffled, never starting with `last`
 * unless `items` has a single element. Used for turn order and for game modes.
 */
export function drawFromBag<T>(bag: readonly T[], items: readonly T[], last: T | null, rng: Rng): BagDraw<T> {
  if (items.length === 0) throw new Error('Cannot draw from a bag without items');
  const valid = bag.filter((item) => items.includes(item));
  const source = valid.length > 0 ? valid : refill(items, last, rng);
  const [item, ...rest] = source;
  if (item === undefined) throw new Error('Shuffle bag produced no item');
  return { item, bag: rest };
}
```

```ts
  return Array.from({ length: count }, (_, index) => card(`${prefix}-${index}`, type, level, text(index)));
}

/** Each call to now() returns a strictly larger time. */
export function steppingClock(start = 1_000, step = 1_000): Clock {
  let current = start;
  return {
    now(): number {
      const value = current;
      current += step;
      return value;
    },
  };
}
```

```ts
import type { Attraction, Gender, Player } from './player';

function attracts(attraction: Attraction, gender: Gender): boolean {
  if (attraction === 'UNDISCLOSED' || gender === 'UNDISCLOSED') return false;
  if (attraction === 'ANYONE') return true;
  if (attraction === 'WOMEN') return gender === 'WOMAN';
  return gender === 'MAN';
}

/** Mutual attraction between two different players. */
export function isCompatible(a: Player, b: Player): boolean {
  if (a.id === b.id) return false;
  return attracts(a.attractedTo, b.gender) && attracts(b.attractedTo, a.gender);
}

export function compatibleWith(player: Player, players: readonly Player[]): Player[] {
  return players.filter((other) => isCompatible(player, other));
}
```

```ts
import type { Card } from '../content/types';
import { extractVariableNames, isKnownVariable, replaceVariables, type Variable } from '../content/variables';
import { compatibleWith } from './matchmaker';
import type { Player } from './player';
import { pickOne, type Rng } from './rng';

/** Most constrained first, so {otro} never steals the only compatible player. */
const RESOLUTION_ORDER: readonly Variable[] = ['jugador', 'compatible', 'otro'];

export interface RenderContext {
  readonly current: Player | undefined;
  readonly players: readonly Player[];
  readonly lastNamed: Readonly<Record<string, number>>;
  readonly rng: Rng;
}

export interface RenderResult {
  readonly text: string;
  /** Players named by {otro} or {compatible}; never the current player. */
  readonly namedPlayerIds: readonly string[];
}

type Chooser = (options: readonly Player[]) => Player;

function variablesIn(card: Card): Set<Variable> {
  const result = new Set<Variable>();
  for (const name of extractVariableNames(card.text)) {
    if (!isKnownVariable(name)) throw new Error(`Card "${card.id}" uses unknown variable {${name}}`);
    result.add(name);
  }
  return result;
}

function candidates(
  variable: Variable,
  current: Player | undefined,
  players: readonly Player[],
  taken: ReadonlySet<string>,
): Player[] {
  if (current === undefined) return [];
  if (variable === 'jugador') return [current];
  const pool = variable === 'compatible' ? compatibleWith(current, players) : players.filter((p) => p.id !== current.id);
  return pool.filter((p) => !taken.has(p.id));
}

function assign(
  card: Card,
  current: Player | undefined,
  players: readonly Player[],
  choose: Chooser,
): Map<Variable, Player> | null {
  const used = variablesIn(card);
  const assignment = new Map<Variable, Player>();
  const taken = new Set<string>();
  for (const variable of RESOLUTION_ORDER) {
    if (!used.has(variable)) continue;
    const options = candidates(variable, current, players, taken);
    if (options.length === 0) return null;
    const chosen = choose(options);
    assignment.set(variable, chosen);
    taken.add(chosen.id);
  }
  return assignment;
}

const firstOption: Chooser = (options) => {
  const first = options[0];
  if (first === undefined) throw new Error('Chooser called without options');
  return first;
};

function leastRecentlyNamed(lastNamed: Readonly<Record<string, number>>, rng: Rng): Chooser {
  return (options) => {
    const score = (p: Player): number => lastNamed[p.id] ?? Number.NEGATIVE_INFINITY;
    const best = Math.min(...options.map(score));
    return pickOne(
      options.filter((p) => score(p) === best),
      rng,
    );
  };
}

export function canRender(card: Card, current: Player | undefined, players: readonly Player[]): boolean {
  return assign(card, current, players, firstOption) !== null;
}

export function renderCard(card: Card, ctx: RenderContext): RenderResult {
  const assignment = assign(card, ctx.current, ctx.players, leastRecentlyNamed(ctx.lastNamed, ctx.rng));
  if (assignment === null) {
    throw new Error(
      `Card "${card.id}" cannot be rendered for ${ctx.current?.name ?? 'the group'} with players [${ctx.players
        .map((p) => p.name)
        .join(', ')}]`,
    );
  }
  const text = replaceVariables(card.text, (name) => {
    const assigned = isKnownVariable(name) ? assignment.get(name) : undefined;
    if (assigned === undefined) throw new Error(`Card "${card.id}" has unresolved variable {${name}}`);
    return assigned.name;
  });
  const namedPlayerIds = [...assignment]
    .filter(([variable]) => variable !== 'jugador')
    .map(([, assigned]) => assigned.id);
  return { text, namedPlayerIds };
}
```

```ts
import type { Card, CardType, Level } from '../content/types';
import { pickOne, type Rng } from './rng';

/** Once every eligible card was seen, draw among this fraction of the oldest ones. */
export const RECYCLE_FRACTION = 0.25;

export interface PickRequest {
  readonly cards: readonly Card[];
  readonly type: CardType;
  readonly level: Level;
  /** cardId -> last time shown (epoch ms). */
  readonly history: Readonly<Record<string, number>>;
  readonly isEligible: (card: Card) => boolean;
  readonly rng: Rng;
}

export function pickCard(request: PickRequest): Card | null {
  const { history, rng } = request;
  const pool = request.cards.filter(
    (card) => card.type === request.type && card.level === request.level && request.isEligible(card),
  );
  if (pool.length === 0) return null;

  const unseen = pool.filter((card) => history[card.id] === undefined);
  if (unseen.length > 0) return pickOne(unseen, rng);

  const oldestFirst = [...pool].sort((a, b) => (history[a.id] ?? 0) - (history[b.id] ?? 0));
  const windowSize = Math.max(1, Math.ceil(pool.length * RECYCLE_FRACTION));
  return pickOne(oldestFirst.slice(0, windowSize), rng);
}
```

Nota para quien revise: el `?? 0` del `sort` no es un valor de rescate. Ese código solo corre cuando `unseen` está vacío, es decir, cuando todas las cartas tienen historial. El `??` solo está para que compile con `noUncheckedIndexedAccess`.

```ts
import { MODE_CARD_TYPES, type Card, type CardType, type GameMode, type Level } from '../content/types';
import { pickCard } from './cardPicker';
import type { Clock } from './clock';
import type { Player } from './player';
import { pickOne, type Rng } from './rng';
import { drawFromBag } from './shuffleBag';
import { canRender, renderCard } from './template';

export const MIN_PLAYERS = 2;

export type Penalty =
  | { readonly kind: 'DRINKS'; readonly count: 1 | 2 | 3 | 4 | 5 }
  | { readonly kind: 'GARMENT' }
  | { readonly kind: 'NONE' };

export interface GameConfig {
  readonly modes: readonly GameMode[];
  readonly level: Level;
  readonly penalty: Penalty;
}

export type PersonalType = 'TRUTH' | 'DARE';
export type Choice = PersonalType | 'RANDOM';
const PERSONAL_TYPES: readonly PersonalType[] = ['TRUTH', 'DARE'];

export type Phase =
  | { readonly kind: 'AWAITING_CHOICE'; readonly playerId: string; readonly unavailable: readonly PersonalType[] }
  | {
      readonly kind: 'SHOWING_CARD';
      readonly mode: GameMode;
      readonly card: Card;
      readonly text: string;
      readonly playerId: string | null;
    }
  | { readonly kind: 'SHOWING_PENALTY'; readonly playerId: string; readonly penalty: Penalty }
  | { readonly kind: 'PAUSED_NEED_PLAYERS' }
  | { readonly kind: 'ENDED' };

export interface ChoiceCount {
  readonly truth: number;
  readonly dare: number;
}

export interface GameState {
  readonly config: GameConfig;
  readonly players: readonly Player[];
  readonly turnQueue: readonly string[];
  readonly lastTurnPlayerId: string | null;
  readonly modeBag: readonly GameMode[];
  readonly lastMode: GameMode | null;
  readonly choiceCount: Readonly<Record<string, ChoiceCount>>;
  /** playerId -> round in which they were last named by {otro} or {compatible}. */
  readonly lastNamed: Readonly<Record<string, number>>;
  /** cardId -> last time shown (epoch ms). Persisted by the UI after every card. */
  readonly history: Readonly<Record<string, number>>;
  readonly round: number;
  readonly phase: Phase;
}

export type GameAction =
  | { readonly type: 'CHOOSE'; readonly choice: Choice }
  | { readonly type: 'COMPLETE' }
  | { readonly type: 'REFUSE' }
  | { readonly type: 'NEXT' }
  | { readonly type: 'SKIP_TURN' }
  | { readonly type: 'END' };

export interface EngineDeps {
  /** Pack cards plus custom cards, every level the UI has loaded. */
  readonly cards: readonly Card[];
  readonly rng: Rng;
  readonly clock: Clock;
}

export class InvalidActionError extends Error {
  constructor(action: GameAction, phase: Phase, reason: string) {
    super(`Action ${action.type} is not valid in phase ${phase.kind}: ${reason}`);
    this.name = 'InvalidActionError';
  }
}

/** Modes that lack at least one of their card types at the configured level. */
export function missingContent(config: GameConfig, cards: readonly Card[]): GameMode[] {
  return config.modes.filter((mode) =>
    MODE_CARD_TYPES[mode].some((type) => !cards.some((card) => card.type === type && card.level === config.level)),
  );
}

function assertContent(config: GameConfig, cards: readonly Card[]): void {
  const missing = missingContent(config, cards);
  if (missing.length > 0) throw new Error(`No cards at level ${config.level} for modes: ${missing.join(', ')}`);
}

function findPlayer(state: GameState, playerId: string): Player {
  const found = state.players.find((p) => p.id === playerId);
  if (found === undefined) throw new Error(`Unknown player id ${playerId}`);
  return found;
}

function unavailableTypes(state: GameState, playerId: string, deps: EngineDeps): PersonalType[] {
  const current = findPlayer(state, playerId);
  return PERSONAL_TYPES.filter(
    (type) =>
      !deps.cards.some(
        (card) => card.type === type && card.level === state.config.level && canRender(card, current, state.players),
      ),
  );
}

function showCard(
  state: GameState,
  mode: GameMode,
  type: CardType,
  playerId: string | null,
  deps: EngineDeps,
): GameState {
  const current = playerId === null ? undefined : findPlayer(state, playerId);
  const card = pickCard({
    cards: deps.cards,
    type,
    level: state.config.level,
    history: state.history,
    isEligible: (candidate) => canRender(candidate, current, state.players),
    rng: deps.rng,
  });
  if (card === null) {
    throw new Error(`No eligible ${type} card at level ${state.config.level} for ${playerId ?? 'the group'}`);
  }
  const rendered = renderCard(card, { current, players: state.players, lastNamed: state.lastNamed, rng: deps.rng });

  const lastNamed: Record<string, number> = { ...state.lastNamed };
  for (const id of rendered.namedPlayerIds) lastNamed[id] = state.round;

  let choiceCount = state.choiceCount;
  if (playerId !== null && (type === 'TRUTH' || type === 'DARE')) {
    const previous = state.choiceCount[playerId] ?? { truth: 0, dare: 0 };
    const updated =
      type === 'TRUTH' ? { ...previous, truth: previous.truth + 1 } : { ...previous, dare: previous.dare + 1 };
    choiceCount = { ...state.choiceCount, [playerId]: updated };
  }

  return {
    ...state,
    history: { ...state.history, [card.id]: deps.clock.now() },
    lastNamed,
    choiceCount,
    phase: { kind: 'SHOWING_CARD', mode, card, text: rendered.text, playerId },
  };
}

function startRound(state: GameState, deps: EngineDeps): GameState {
  const modeDraw = drawFromBag(state.modeBag, state.config.modes, state.lastMode, deps.rng);
  const next: GameState = { ...state, round: state.round + 1, modeBag: modeDraw.bag, lastMode: modeDraw.item };

  if (modeDraw.item === 'TRUTH_OR_DARE') {
    const turn = drawFromBag(
      state.turnQueue,
      state.players.map((p) => p.id),
      state.lastTurnPlayerId,
      deps.rng,
    );
    return {
      ...next,
      turnQueue: turn.bag,
      lastTurnPlayerId: turn.item,
      phase: { kind: 'AWAITING_CHOICE', playerId: turn.item, unavailable: unavailableTypes(next, turn.item, deps) },
    };
  }

  const [groupType] = MODE_CARD_TYPES[modeDraw.item];
  if (groupType === undefined) throw new Error(`Mode ${modeDraw.item} has no card types`);
  return showCard(next, modeDraw.item, groupType, null, deps);
}

function resolveChoice(
  state: GameState,
  phase: Extract<Phase, { kind: 'AWAITING_CHOICE' }>,
  action: Extract<GameAction, { type: 'CHOOSE' }>,
  rng: Rng,
): PersonalType {
  const available = PERSONAL_TYPES.filter((type) => !phase.unavailable.includes(type));
  const choice = action.choice;
  if (choice !== 'RANDOM') {
    if (!available.includes(choice)) {
      throw new InvalidActionError(action, phase, `no ${choice} card is possible for ${phase.playerId}`);
    }
    return choice;
  }
  const [first, ...others] = available;
  if (first === undefined) throw new InvalidActionError(action, phase, `no card is possible for ${phase.playerId}`);
  if (others.length === 0) return first;
  const count = state.choiceCount[phase.playerId] ?? { truth: 0, dare: 0 };
  if (count.truth < count.dare) return 'TRUTH';
  if (count.dare < count.truth) return 'DARE';
  return pickOne(PERSONAL_TYPES, rng);
}

function assertNever(value: never): never {
  throw new Error(`Unhandled action: ${JSON.stringify(value)}`);
}

export function createGame(
  config: GameConfig,
  players: readonly Player[],
  history: Readonly<Record<string, number>>,
  deps: EngineDeps,
): GameState {
  if (config.modes.length === 0) throw new Error('A game needs at least one mode');
  if (new Set(config.modes).size !== config.modes.length) {
    throw new Error(`Duplicate modes: ${config.modes.join(', ')}`);
  }
  if (players.length < MIN_PLAYERS) {
    throw new Error(`A game needs at least ${MIN_PLAYERS} players, got ${players.length}`);
  }
  const ids = players.map((p) => p.id);
  if (new Set(ids).size !== ids.length) throw new Error(`Duplicate player ids: ${ids.join(', ')}`);
  assertContent(config, deps.cards);

  const initial: GameState = {
    config,
    players,
    turnQueue: [],
    lastTurnPlayerId: null,
    modeBag: [],
    lastMode: null,
    choiceCount: {},
    lastNamed: {},
    history,
    round: 0,
    phase: { kind: 'PAUSED_NEED_PLAYERS' },
  };
  return startRound(initial, deps);
}

export function reduce(state: GameState, action: GameAction, deps: EngineDeps): GameState {
  const { phase } = state;
  if (phase.kind === 'ENDED') throw new InvalidActionError(action, phase, 'the game has ended');

  switch (action.type) {
    case 'END':
      return { ...state, phase: { kind: 'ENDED' } };

    case 'CHOOSE': {
      if (phase.kind !== 'AWAITING_CHOICE') throw new InvalidActionError(action, phase, 'no player is choosing');
      const type = resolveChoice(state, phase, action, deps.rng);
      return showCard(state, 'TRUTH_OR_DARE', type, phase.playerId, deps);
    }

    case 'COMPLETE': {
      if (phase.kind !== 'SHOWING_CARD' || phase.playerId === null) {
        throw new InvalidActionError(action, phase, 'only a personal card can be completed');
      }
      return startRound(state, deps);
    }

    case 'REFUSE': {
      if (phase.kind !== 'SHOWING_CARD' || phase.playerId === null) {
        throw new InvalidActionError(action, phase, 'only a personal card can be refused');
      }
      if (state.config.penalty.kind === 'NONE') return startRound(state, deps);
      return { ...state, phase: { kind: 'SHOWING_PENALTY', playerId: phase.playerId, penalty: state.config.penalty } };
    }

    case 'NEXT': {
      const isGroupCard = phase.kind === 'SHOWING_CARD' && phase.playerId === null;
      if (!isGroupCard && phase.kind !== 'SHOWING_PENALTY') {
        throw new InvalidActionError(action, phase, 'NEXT only follows a group card or a penalty');
      }
      return startRound(state, deps);
    }

    case 'SKIP_TURN': {
      if (phase.kind !== 'AWAITING_CHOICE' || phase.unavailable.length < PERSONAL_TYPES.length) {
        throw new InvalidActionError(action, phase, 'a turn can only be skipped when no card is possible');
      }
      return startRound(state, deps);
    }

    default:
      return assertNever(action);
  }
}
```

```ts
import { describe, expect, it } from 'vitest';
import { GAME_MODES, type Card, type CardType } from '../content/types';
import { deck, player, steppingClock } from '../testing/fixtures';
import { createGame, reduce, type EngineDeps, type GameState } from './engine';
import { createSeededRng } from './rng';
import { canRender } from './template';

const players = [
  player('ana', 'WOMAN', 'MEN'),
  player('beto', 'MAN', 'WOMEN'),
  player('caro', 'WOMAN', 'ANYONE'),
  player('dani', 'OTHER', 'UNDISCLOSED'),
];

const PER_TYPE = 30;
const cards: Card[] = [
  ...deck('t', 'TRUTH', 'SOFT', PER_TYPE, (i) => (i % 3 === 0 ? `¿Qué opinas de {otro}? (${i})` : `Verdad ${i}`)),
  ...deck('d', 'DARE', 'SOFT', PER_TYPE, (i) => (i % 4 === 0 ? `Choca la mano de {compatible} (${i})` : `Reto ${i}`)),
  ...deck('n', 'NEVER', 'SOFT', PER_TYPE),
  ...deck('r', 'RATHER', 'SOFT', PER_TYPE),
  ...deck('m', 'MOST_LIKELY', 'SOFT', PER_TYPE),
];

interface Shown {
  readonly card: Card;
  readonly playerId: string | null;
}

interface Transcript {
  readonly turns: string[];
  readonly shown: Shown[];
  readonly texts: string[];
  readonly final: GameState;
}

function playRounds(seed: number, rounds: number): Transcript {
  const d: EngineDeps = { cards, rng: createSeededRng(seed), clock: steppingClock() };
  let state = createGame({ modes: [...GAME_MODES], level: 'SOFT', penalty: { kind: 'NONE' } }, players, {}, d);
  const turns: string[] = [];
  const shown: Shown[] = [];
  const texts: string[] = [];
  while (state.round <= rounds) {
    const { phase } = state;
    if (phase.kind === 'AWAITING_CHOICE') {
      turns.push(phase.playerId);
      state = reduce(state, { type: 'CHOOSE', choice: 'RANDOM' }, d);
    } else if (phase.kind === 'SHOWING_CARD') {
      shown.push({ card: phase.card, playerId: phase.playerId });
      texts.push(phase.text);
      state = reduce(state, phase.playerId === null ? { type: 'NEXT' } : { type: 'COMPLETE' }, d);
    } else {
      throw new Error(`Unexpected phase ${phase.kind} in round ${state.round}`);
    }
  }
  return { turns, shown, texts, final: state };
}

describe('long game', () => {
  const transcript = playRounds(42, 300);

  it('never gives two truth-or-dare turns in a row to the same player', () => {
    for (let i = 1; i < transcript.turns.length; i++) expect(transcript.turns[i]).not.toBe(transcript.turns[i - 1]);
  });

  it('gives everyone exactly one turn per cycle', () => {
    const ids = players.map((p) => p.id).sort();
    const complete = transcript.turns.length - (transcript.turns.length % ids.length);
    for (let start = 0; start < complete; start += ids.length) {
      expect(transcript.turns.slice(start, start + ids.length).sort()).toEqual(ids);
    }
  });

  it('repeats a card only when no unseen card was eligible for that turn', () => {
    const seen = new Set<string>();
    let repeats = 0;
    for (const { card: shownCard, playerId } of transcript.shown) {
      if (seen.has(shownCard.id)) {
        repeats++;
        const current = playerId === null ? undefined : players.find((p) => p.id === playerId);
        const unseenEligible = cards.filter(
          (other) =>
            other.type === shownCard.type &&
            other.level === shownCard.level &&
            !seen.has(other.id) &&
            canRender(other, current, players),
        );
        expect(unseenEligible.map((c) => c.id), `${shownCard.id} repeated while unseen eligible cards remained`).toEqual([]);
      }
      seen.add(shownCard.id);
    }
    expect(repeats, 'the game should be long enough to exercise recycling').toBeGreaterThan(0);
  });

  it('shows every card of a type at least once before the first repeat when all are eligible for everyone', () => {
    const always: CardType[] = ['NEVER', 'RATHER', 'MOST_LIKELY'];
    for (const type of always) {
      const ids = transcript.shown.filter((s) => s.card.type === type).map((s) => s.card.id);
      const firstCycle = ids.slice(0, PER_TYPE);
      expect(new Set(firstCycle).size, `${type} repeated before exhausting its pool`).toBe(firstCycle.length);
    }
  });

  it('keeps truths and dares balanced per player when everyone picks RANDOM', () => {
    for (const count of Object.values(transcript.final.choiceCount)) {
      expect(Math.abs(count.truth - count.dare)).toBeLessThanOrEqual(1);
    }
  });

  it('never leaves template variables unresolved', () => {
    for (const text of transcript.texts) expect(text).not.toMatch(/[{}]/);
  });

  it('is reproducible with the same seed and different with another', () => {
    expect(playRounds(42, 60).texts).toEqual(playRounds(42, 60).texts);
    expect(playRounds(42, 60).texts).not.toEqual(playRounds(43, 60).texts);
  });
});
```
